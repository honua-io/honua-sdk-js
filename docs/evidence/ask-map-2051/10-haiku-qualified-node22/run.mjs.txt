async function invoke(_url, options) {
  const start = Date.now();
  writeFileSync('/tmp/ask2051-live-provider/last-provider-request.json', options.body, {mode:0o600});
  return await new Promise((resolve, reject) => {
    const child = spawn('python3', ['/tmp/ask2051-live-provider/invoke.py'], { signal: options.signal, stdio: ['pipe', 'pipe', 'pipe'] });
    let stdout = '';
    let stderr = '';
    child.stdout.on('data', chunk => { stdout += chunk; if (stdout.length > 65536) child.kill(); });
    child.stderr.on('data', chunk => { stderr += chunk; });
    child.on('error', reject);
    child.on('close', code => {
      if (code !== 0) {
        calls.push({ elapsedMs: Date.now() - start, outcome: 'provider-error', providerError: (() => { try { return JSON.parse(stderr); } catch { return {error:'provider-failed'}; } })() });
        reject(new Error('Bedrock invocation failed'));
        return;
      }
      try {
        const body = JSON.parse(stdout);
        calls.push({ requestId: body.providerRequestId, model: body.model, elapsedMs: Date.now() - start, usage: body.usage, toolCalls: body.choices[0].message.tool_calls.map(call => ({name:call.function.name, arguments:JSON.parse(call.function.arguments)})), outcome: 'provider-response' });
        resolve(new Response(JSON.stringify(body), { status: 200 }));
      } catch(error) { reject(error); }
    });
    child.stdin.end(options.body);
  });
}
